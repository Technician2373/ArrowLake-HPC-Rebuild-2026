 #!/usr/bin/env bash
set -e

echo "Updating system..."
sudo apt update -y && sudo apt full-upgrade -y
sleep=3

###############################################
# CORE BUILD TOOLCHAIN
###############################################
echo "Installing core build tools..."
sudo apt install -y \
    build-essential cmake make pkg-config \
    gcc g++ gdb ltrace strace \
    libncurses-dev libelf-dev libssl-dev libdw-dev dwarves \
    bison flex libcap-dev libpcre2-dev
sleep=3

###############################################
# KERNEL DEVELOPMENT + DEBUGGING
###############################################
echo "Installing kernel dev + debugging tools..."
sudo apt install -y \
    linux-headers-$(uname -r) \
    pahole \
    systemtap \
    crash \
    kexec-tools \
    elfutils \
    libtraceevent-dev \
    libtracefs-dev
sleep=3

###############################################
# NETWORKING + DIAGNOSTICS
###############################################
echo "Installing networking + diagnostics..."
sudo apt install -y \
    traceroute nmap wireshark \
    net-tools wavemon \
    iw rfkill \
    iftop nvtop iotop atop htop btop \
    tcpdump mtr iperf3 ethtool
sleep=3

###############################################
# FILESYSTEM + DISK TOOLS
###############################################
echo "Installing filesystem + disk tools..."
sudo apt install -y \
    gparted parted gdisk fdisk \
    dosfstools cryptsetup lvm2 \
    efibootmgr sbsigntool mokutil \
    nvme-cli rsync tar \
    exfatprogs btrfs-progs xfsprogs
sleep=3

###############################################
# DEVELOPMENT ENVIRONMENT
###############################################
echo "Installing developer environment..."
sudo apt install -y \
    git wget curl fastfetch \
    python3 python3-pip python3-venv \
    nodejs npm \
    vulkan-tools clinfo \
    neovim vim micro fzf bat tree locate \
    jq ripgrep
sleep=3

###############################################
# SYSTEM UTILITIES + SECURITY
###############################################
echo "Installing system utilities..."
sudo apt install -y \
    fail2ban timeshift snapd \
    ufw apparmor-utils \
    lm-sensors powertop
sleep=3

###############################################
# VIRTUALIZATION + CONTAINERS
###############################################
echo "Installing virtualization tools..."
sudo apt install -y \
    qemu-system-x86 qemu-utils \                                                            
    libvirt-daemon-system libvirt-clients \
    virt-manager bridge-utils \
    docker.io docker-compose
sleep=3

###############################################
# OPTIONAL: PERFORMANCE + BENCHMARKING
###############################################
echo "Installing performance tools..."
sudo apt install -y \
    stress-ng sysbench \
    glmark2 mesa-utils \
    perf-tools-unstable
sleep=5

echo "Done. All dependencies installed."
